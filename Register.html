function registerUser(event) { event.preventDefault(); const password = document.getElementById("password").value; const confirmPassword = document.getElementById("confirm").value; const msg = document.getElementById("msg"); if (password.length < 6) { msg.textContent = "Password must be at least 6 characters."; return; } if (password !== confirmPassword) { msg.textContent = "Passwords do not match!"; return; } // Redirect to index.html after validation window.location.href = "index.html"; }
